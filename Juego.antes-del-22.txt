"use client";

import { useEffect, useState } from "react";
import Escena from "./Escena";
import Menu from "./Menu";
import { PERFIL_INICIAL, type Perfil } from "@/lib/perfil";

const CLAVE = "apex-perfil";

export default function Juego() {
  const [fase, setFase] = useState<"menu" | "carrera">("menu");
  const [perfil, setPerfil] = useState<Perfil>(PERFIL_INICIAL);

  useEffect(() => {
    try {
      const guardado = localStorage.getItem(CLAVE);
      if (guardado) setPerfil({ ...PERFIL_INICIAL, ...JSON.parse(guardado) });
    } catch {}
  }, []);

  const cambiar = (p: Perfil) => {
    setPerfil(p);
    try {
      localStorage.setItem(CLAVE, JSON.stringify(p));
    } catch {}
  };

  if (fase === "menu") {
    return <Menu perfil={perfil} alCambiar={cambiar} alJugar={() => setFase("carrera")} />;
  }

  return (
    <div className="relative h-screen w-screen overflow-hidden">
      <Escena perfil={perfil} />

      <div className="pointer-events-none absolute left-5 top-5 flex items-center gap-3 rounded-full border border-white/10 bg-black/35 py-2 pl-3 pr-5 backdrop-blur-md">
        <span
          className="h-3 w-3 rounded-full"
          style={{ backgroundColor: perfil.color, boxShadow: `0 0 12px ${perfil.color}` }}
        />
        <span className="font-mono text-xs uppercase tracking-[0.2em]">{perfil.nombre}</span>
      </div>

      <button
        type="button"
        onClick={() => setFase("menu")}
        className="absolute right-5 top-5 rounded-full border border-white/10 bg-black/35 px-5 py-2 font-mono text-xs uppercase tracking-[0.2em] backdrop-blur-md transition hover:bg-black/55"
      >
        Menú
      </button>

      <p className="anim-hint pointer-events-none absolute bottom-8 left-1/2 -translate-x-1/2 font-mono text-[11px] uppercase tracking-[0.3em] text-white/70">
        A · D o ← → para cambiar de carril
      </p>
    </div>
  );
}
